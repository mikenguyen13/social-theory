% ---- Running headers ------------------------------------------------------------
% LaTeX marks are lost when a long table starts on a chapter's first page, which
% left stale chapter titles in the header. Store the title in a macro instead:
% \chapter clears the page first, so every page of a chapter gets its own title.
\usepackage{fancyhdr}
\newcommand{\currentchapterhead}{}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\footnotesize\sffamily\color{stgrey}\MakeUppercase{\currentchapterhead}}
\fancyhead[R]{\footnotesize\sffamily\thepage}
\renewcommand{\headrulewidth}{0.3pt}
\fancypagestyle{plain}{\fancyhf{}\fancyfoot[C]{\footnotesize\sffamily\thepage}\renewcommand{\headrulewidth}{0pt}}
\AtBeginDocument{%
  \renewcommand{\chaptermark}[1]{\gdef\currentchapterhead{\chaptername\ \thechapter.\ #1}}%
  \renewcommand{\sectionmark}[1]{}}
\newcommand{\setchapterhead}[1]{\gdef\currentchapterhead{#1}}
